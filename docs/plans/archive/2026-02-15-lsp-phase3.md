### Task 1: Rename — prepareRename

**Files:**
- Modify: `src/lsp/mod.rs`

The `prepareRename` handler validates the cursor is on a renameable element and returns its current text + range. Two contexts:

1. **Cursor on a wikilink** → rename the linked page (target_raw is the placeholder).
2. **Cursor on the frontmatter title line** → rename the current page.

For frontmatter title detection: if the cursor position converts to `None` via `position_to_body_byte_offset` (i.e. it's before the body), check the line text for a `title:` prefix.

**Step 1: Implement `prepare_rename` on `LanguageServer`**

Add to the `LanguageServer` impl in `src/lsp/mod.rs`:

```rust
async fn prepare_rename(
    &self,
    params: TextDocumentPositionParams,
) -> Result<Option<PrepareRenameResponse>> {
    let uri = params.text_document.uri;
    let pos = params.position;

    let docs = self.documents.lock().await;
    let doc = match docs.get(&uri) {
        Some(d) => d,
        None => return Ok(None),
    };

    // 1. Check if cursor is on a wikilink
    if let Some(link) = doc.link_at_position(pos) {
        if link.kind == crate::vault::link::LinkKind::Wiki {
            let range = doc.link_to_range(link);
            return Ok(Some(PrepareRenameResponse::RangeWithPlaceholder {
                range,
                placeholder: link.target_raw.clone(),
            }));
        }
    }

    // 2. Check if cursor is on the frontmatter title
    if doc.position_to_body_byte_offset(pos).is_none() {
        // Cursor is in frontmatter region
        let line_idx = pos.line as usize;
        if line_idx < doc.rope.len_lines() {
            let line_text = doc.rope.line(line_idx).to_string();
            if let Some(title_value) = line_text.strip_prefix("title:") {
                let title = title_value.trim().trim_matches('"').trim_matches('\'');
                if !title.is_empty() {
                    let title_start_byte = {
                        let line_start = doc.rope.line_to_byte(line_idx);
                        // Find where the value starts (after "title:" and whitespace)
                        let value_offset = line_text.len() - line_text.trim_start_matches("title:").len();
                        let ws_after = line_text[value_offset..].len()
                            - line_text[value_offset..].trim_start().len();
                        line_start + value_offset + ws_after
                    };
                    // Trim quotes from the range
                    let raw_value = title_value.trim();
                    let (adj_start, adj_len) = if (raw_value.starts_with('"') && raw_value.ends_with('"'))
                        || (raw_value.starts_with('\'') && raw_value.ends_with('\''))
                    {
                        (title_start_byte + 1, raw_value.len() - 2)
                    } else {
                        (title_start_byte, raw_value.len())
                    };

                    let start_line = doc.rope.byte_to_line(adj_start);
                    let start_char = adj_start - doc.rope.line_to_byte(start_line);
                    let end_byte = adj_start + adj_len;
                    let end_line = doc.rope.byte_to_line(end_byte);
                    let end_char = end_byte - doc.rope.line_to_byte(end_line);

                    return Ok(Some(PrepareRenameResponse::RangeWithPlaceholder {
                        range: Range {
                            start: Position { line: start_line as u32, character: start_char as u32 },
                            end: Position { line: end_line as u32, character: end_char as u32 },
                        },
                        placeholder: title.to_string(),
                    }));
                }
            }
        }
    }

    Ok(None)
}
```

This is the core rename handler. It computes a `WorkspaceEdit` with:

```rust
use crate::vault::canonical::CanonicalName;

/// Given the raw source text at a wikilink span (e.g. `[[Old Name]]` or
/// `[[Old Name|Custom Display]]`), compute the replacement text with a new target.
///
/// Preserves display text if present: `[[Old|Show]]` → `[[New|Show]]`.
pub fn rewrite_wikilink(raw_span_text: &str, new_target: &str) -> String {
    // Strip outer brackets
    let inner = raw_span_text
        .strip_prefix("[[")
        .and_then(|s| s.strip_suffix("]]"))
        .unwrap_or(raw_span_text);

    if let Some((_old_target, display)) = inner.split_once('|') {
        format!("[[{new_target}|{display}]]")
    } else {
        format!("[[{new_target}]]")
    }
}

/// Check whether a link's target resolves to any of the given canonical names.
pub fn link_matches_target(target_raw: &str, target_canonical_names: &[String]) -> bool {
    let cn = CanonicalName::from_title(target_raw);
    target_canonical_names.iter().any(|name| name == cn.as_str())
}

/// Compute the new frontmatter content after updating the title field.
///
/// Replaces the `title: ...` line in the YAML frontmatter with the new title.
/// Returns the full document text with the updated frontmatter.
pub fn update_frontmatter_title(full_text: &str, new_title: &str) -> Option<String> {
    let (meta, body) = crate::vault::page::parse_frontmatter(full_text).ok()?;
    let mut new_meta = meta;
    new_meta.title = Some(new_title.to_string());
    Some(crate::vault::page::write_page_content(&new_meta, &body))
}
```

**Step 3: Implement `rename` on `LanguageServer`**

Add to the `LanguageServer` impl in `src/lsp/mod.rs`:

```rust
async fn rename(&self, params: RenameParams) -> Result<Option<WorkspaceEdit>> {
    let uri = params.text_document_position.text_document.uri;
    let pos = params.text_document_position.position;
    let new_name = params.new_name;

    // Determine what's being renamed and resolve the old target
    let (old_vp, old_title) = {
        let docs = self.documents.lock().await;
        let doc = match docs.get(&uri) {
            Some(d) => d,
            None => return Ok(None),
        };

        // Check wikilink first
        if let Some(link) = doc.link_at_position(pos) {
            if link.kind == crate::vault::link::LinkKind::Wiki {
                // Resolve the target to a vault path
                let canonical = crate::vault::canonical::CanonicalName::from_title(&link.target_raw);
                let info: Option<(String, Option<String>)> = self
                    .state
                    .index
                    .with_index({
                        let cn = canonical.as_str().to_string();
                        move |index, _| -> std::result::Result<_, rusqlite::Error> {
                            index.connection().query_row(
                                "SELECT p.path, p.title FROM canonical_names cn \
                                 JOIN pages p ON p.id = cn.page_id \
                                 WHERE cn.canonical_name = ?1 LIMIT 1",
                                rusqlite::params![cn],
                                |row| Ok((row.get::<_, String>(0)?, row.get::<_, Option<String>>(1)?)),
                            )
                        }
                    })
                    .await
                    .map_err(|_| tower_lsp::jsonrpc::Error::internal_error())?
                    .ok();

                match info {
                    Some((path, title)) => {
                        let vp = crate::vault::path::VaultPath::new(&path)
                            .map_err(|_| tower_lsp::jsonrpc::Error::internal_error())?;
                        (vp, title.unwrap_or_else(|| vp.stem().to_string()))
                    }
                    None => return Ok(None), // unresolved link — can't rename
                }
            } else {
                return Ok(None); // non-wiki link
            }
        } else if doc.position_to_body_byte_offset(pos).is_none() {
            // Frontmatter title rename — target is current page
            let vp = match self.uri_to_vault_path(&uri) {
                Some(vp) => vp,
                None => return Ok(None),
            };
            let title = doc.meta.title.clone().unwrap_or_else(|| vp.stem().to_string());
            (vp, title)
        } else {
            return Ok(None);
        }
    };

    // Compute new vault path (keep same parent folder)
    let parent = old_vp.parent().unwrap_or("").to_string();
    let new_filename = crate::vault::path::VaultPath::from_title(&new_name);
    let new_path_str = if parent.is_empty() {
        new_filename.as_str().to_string()
    } else {
        format!("{}/{}", parent, new_filename.as_str())
    };
    let new_vp = crate::vault::path::VaultPath::new(&new_path_str)
        .map_err(|_| tower_lsp::jsonrpc::Error::internal_error())?;

    // Check for conflict
    let new_abs = self.state.vault.resolve(&new_vp);
    if new_abs.exists() && new_vp.as_str() != old_vp.as_str() {
        return Err(tower_lsp::jsonrpc::Error {
            code: tower_lsp::jsonrpc::ErrorCode::InvalidParams,
            message: format!("Target path already exists: {}", new_vp.as_str()).into(),
            data: None,
        });
    }

    // Get all canonical names for the old page
    let old_canonical_names: Vec<String> = self
        .state
        .index
        .with_index({
            let old_path = old_vp.as_str().to_string();
            move |index, _| -> std::result::Result<Vec<String>, rusqlite::Error> {
                let page_id: String = index.connection().query_row(
                    "SELECT id FROM pages WHERE path = ?1",
                    rusqlite::params![old_path],
                    |row| row.get(0),
                )?;
                let mut stmt = index.connection().prepare(
                    "SELECT canonical_name FROM canonical_names WHERE page_id = ?1",
                )?;
                let names = stmt
                    .query_map(rusqlite::params![page_id], |row| row.get::<_, String>(0))?
                    .filter_map(|r| r.ok())
                    .collect();
                Ok(names)
            }
        })
        .await
        .map_err(|_| tower_lsp::jsonrpc::Error::internal_error())?
        .unwrap_or_default();

    if old_canonical_names.is_empty() {
        return Ok(None);
    }

    // Find all referring pages via reverse_deps
    let referring_paths: Vec<String> = self
        .state
        .index
        .with_index({
            let old_path = old_vp.as_str().to_string();
            let cnames = old_canonical_names.clone();
            move |index, _| -> std::result::Result<Vec<String>, rusqlite::Error> {
                // Get page_id
                let page_id: Option<String> = index.connection().query_row(
                    "SELECT id FROM pages WHERE path = ?1",
                    rusqlite::params![old_path],
                    |row| row.get(0),
                ).ok();

                let mut paths = std::collections::HashSet::new();

                // Resolved links (target_id matches)
                if let Some(ref pid) = page_id {
                    let mut stmt = index.connection().prepare(
                        "SELECT DISTINCT p.path FROM links l \
                         JOIN pages p ON l.source_id = p.id \
                         WHERE l.target_id = ?1"
                    )?;
                    let rows = stmt.query_map(rusqlite::params![pid], |row| row.get::<_, String>(0))?;
                    for r in rows.flatten() {
                        paths.insert(r);
                    }
                }

                // Unresolved links matching canonical names
                for cn in &cnames {
                    let mut stmt = index.connection().prepare(
                        "SELECT DISTINCT p.path FROM links l \
                         JOIN pages p ON l.source_id = p.id \
                         WHERE l.target_canonical = ?1"
                    )?;
                    let rows = stmt.query_map(rusqlite::params![cn], |row| row.get::<_, String>(0))?;
                    for r in rows.flatten() {
                        paths.insert(r);
                    }
                }

                Ok(paths.into_iter().collect())
            }
        })
        .await
        .map_err(|_| tower_lsp::jsonrpc::Error::internal_error())?
        .unwrap_or_default();

    // Build document changes
    let vault_root = self.state.vault.root().to_path_buf();
    let mut ops: Vec<DocumentChangeOperation> = Vec::new();

    // 1. RenameFile if path changed
    let old_abs = self.state.vault.resolve(&old_vp);
    if new_vp.as_str() != old_vp.as_str() {
        let old_uri = Url::from_file_path(&old_abs)
            .map_err(|_| tower_lsp::jsonrpc::Error::internal_error())?;
        let new_uri = Url::from_file_path(&new_abs)
            .map_err(|_| tower_lsp::jsonrpc::Error::internal_error())?;
        ops.push(DocumentChangeOperation::Op(ResourceOp::Rename(RenameFile {
            old_uri,
            new_uri,
            options: Some(RenameFileOptions {
                overwrite: Some(false),
                ignore_if_exists: Some(false),
            }),
            annotation_id: None,
        })));
    }

    // 2. Update frontmatter title on target page
    {
        let target_uri = Url::from_file_path(&old_abs)
            .map_err(|_| tower_lsp::jsonrpc::Error::internal_error())?;

        let target_text = {
            let docs = self.documents.lock().await;
            if let Some(doc) = docs.get(&target_uri) {
                doc.rope.to_string()
            } else {
                tokio::fs::read_to_string(&old_abs).await.unwrap_or_default()
            }
        };

        if let Some(new_content) = rename::update_frontmatter_title(&target_text, &new_name) {
            // Full document replacement edit
            let doc_for_range = document::Document::from_text(&target_text, 0);
            let end_pos = doc_for_range.byte_offset_to_position(
                target_text.len().saturating_sub(doc_for_range.body_byte_offset)
            );
            // Actually compute end from the rope directly
            let last_line = doc_for_range.rope.len_lines().saturating_sub(1);
            let last_line_len = doc_for_range.rope.line(last_line).len_bytes();
            let end = Position {
                line: last_line as u32,
                character: last_line_len as u32,
            };

            ops.push(DocumentChangeOperation::Edit(TextDocumentEdit {
                text_document: OptionalVersionedTextDocumentIdentifier {
                    uri: target_uri,
                    version: None,
                },
                edits: vec![OneOf::Left(TextEdit {
                    range: Range {
                        start: Position { line: 0, character: 0 },
                        end,
                    },
                    new_text: new_content,
                })],
            }));
        }
    }

    // 3. Rewrite wikilinks in each referring page
    for ref_path in &referring_paths {
        // Skip the target page itself (already handled above)
        if ref_path == old_vp.as_str() {
            continue;
        }

        let ref_vp = match crate::vault::path::VaultPath::new(ref_path) {
            Ok(vp) => vp,
            Err(_) => continue,
        };
        let ref_abs = vault_root.join(ref_vp.as_str());
        let ref_uri = match Url::from_file_path(&ref_abs) {
            Ok(u) => u,
            Err(_) => continue,
        };

        // Get document text and links
        let (text, links) = {
            let docs = self.documents.lock().await;
            if let Some(doc) = docs.get(&ref_uri) {
                (doc.rope.to_string(), doc.links.clone())
            } else {
                match tokio::fs::read_to_string(&ref_abs).await {
                    Ok(content) => {
                        let doc = document::Document::from_text(&content, 0);
                        (content, doc.links)
                    }
                    Err(_) => continue,
                }
            }
        };

        let doc = document::Document::from_text(&text, 0);
        let mut edits: Vec<OneOf<TextEdit, AnnotatedTextEdit>> = Vec::new();

        for link in &links {
            if link.kind != crate::vault::link::LinkKind::Wiki {
                continue;
            }
            if link.span.start == 0 && link.span.end == 0 {
                continue; // skip property ref links
            }
            if !rename::link_matches_target(&link.target_raw, &old_canonical_names) {
                continue;
            }

            // Read the raw wikilink text from the source
            let raw_text = if link.span.end <= doc.body.len() {
                &doc.body[link.span.clone()]
            } else {
                continue;
            };
            let new_text = rename::rewrite_wikilink(raw_text, &new_name);
            let range = doc.link_to_range(link);

            edits.push(OneOf::Left(TextEdit {
                range,
                new_text,
            }));
        }

        if !edits.is_empty() {
            ops.push(DocumentChangeOperation::Edit(TextDocumentEdit {
                text_document: OptionalVersionedTextDocumentIdentifier {
                    uri: ref_uri,
                    version: None,
                },
                edits,
            }));
        }
    }

    Ok(Some(WorkspaceEdit {
        changes: None,
        document_changes: Some(DocumentChanges::Operations(ops)),
        change_annotations: None,
    }))
}
```

```rust
async fn code_action(&self, params: CodeActionParams) -> Result<Option<CodeActionResponse>> {
    let uri = params.text_document.uri;
    let mut actions: Vec<CodeActionOrCommand> = Vec::new();

    let docs = self.documents.lock().await;
    let doc = match docs.get(&uri) {
        Some(d) => d,
        None => return Ok(None),
    };

    for diag in &params.context.diagnostics {
        let code = match &diag.code {
            Some(NumberOrString::String(s)) => s.as_str(),
            _ => continue,
        };

        match code {
            "unresolved-link" => {
                // Find the link at this diagnostic range
                if let Some(link) = doc.links.iter().find(|l| {
                    l.kind == crate::vault::link::LinkKind::Wiki
                        && doc.link_to_range(l) == diag.range
                }) {
                    let target = &link.target_raw;
                    let new_vp = crate::vault::path::VaultPath::from_title(target);
                    let new_abs = self.state.vault.resolve(&new_vp);
                    let new_uri = match Url::from_file_path(&new_abs) {
                        Ok(u) => u,
                        Err(_) => continue,
                    };

                    // Build frontmatter scaffold
                    let mut meta = crate::vault::page::PageMeta::new();
                    meta.title = Some(target.to_string());
                    let content = crate::vault::page::write_page_content(&meta, "\n");

                    let mut ops: Vec<DocumentChangeOperation> = Vec::new();
                    ops.push(DocumentChangeOperation::Op(ResourceOp::Create(
                        CreateFile {
                            uri: new_uri.clone(),
                            options: Some(CreateFileOptions {
                                overwrite: Some(false),
                                ignore_if_exists: Some(true),
                            }),
                            annotation_id: None,
                        },
                    )));
                    ops.push(DocumentChangeOperation::Edit(TextDocumentEdit {
                        text_document: OptionalVersionedTextDocumentIdentifier {
                            uri: new_uri,
                            version: None,
                        },
                        edits: vec![OneOf::Left(TextEdit {
                            range: Range::default(),
                            new_text: content,
                        })],
                    }));

                    actions.push(CodeActionOrCommand::CodeAction(CodeAction {
                        title: format!("Create page: {target}"),
                        kind: Some(CodeActionKind::QUICKFIX),
                        diagnostics: Some(vec![diag.clone()]),
                        edit: Some(WorkspaceEdit {
                            changes: None,
                            document_changes: Some(DocumentChanges::Operations(ops)),
                            change_annotations: None,
                        }),
                        is_preferred: Some(true),
                        ..Default::default()
                    }));
                }
            }
            _ => {}
        }
    }

    if actions.is_empty() {
        Ok(None)
    } else {
        Ok(Some(actions))
    }
}
```

```rust
async fn refresh_canonical_names(&self) {
    let result = self
        .state
        .index
        .with_index(|index, _| -> std::result::Result<HashMap<String, Vec<String>>, rusqlite::Error> {
            let mut stmt = index.connection().prepare(
                "SELECT cn.canonical_name, p.path \
                 FROM canonical_names cn \
                 JOIN pages p ON p.id = cn.page_id \
                 ORDER BY cn.canonical_name",
            )?;
            let mut map: HashMap<String, Vec<String>> = HashMap::new();
            let rows = stmt.query_map([], |row| {
                Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?))
            })?;
            for row in rows.flatten() {
                map.entry(row.0).or_default().push(row.1);
            }
            Ok(map)
        })
        .await;

    match result {
        Ok(Ok(names)) => *self.canonical_names.write().await = names,
        Ok(Err(e)) => tracing::error!("failed to load canonical names: {e}"),
        Err(e) => tracing::error!("index thread error loading canonical names: {e}"),
    }
```

```rust
async fn publish_diagnostics_for(&self, uri: &Url, doc: &document::Document) {
    let mut diagnostics = Vec::new();
    let names = self.canonical_names.read().await;

    for link in &doc.links {
        if link.span.start == 0 && link.span.end == 0 {
            continue; // skip property ref links

        let canonical = crate::vault::canonical::CanonicalName::from_title(&link.target_raw);
        let cn_str = canonical.as_str();

        match names.get(cn_str) {
            None => {
                // Unresolved link
                diagnostics.push(Diagnostic {
                    range: doc.link_to_range(link),
                    severity: Some(DiagnosticSeverity::WARNING),

                    message: format!("Unresolved link: \"{}\"", link.target_raw),
                    ..Default::default()
                });
            }
            Some(paths) if paths.len() > 1 => {
                // Ambiguous link
                let vault_root = self.state.vault.root();
                let related: Vec<DiagnosticRelatedInformation> = paths
                    .iter()
                    .filter_map(|p| {
                        let vp = crate::vault::path::VaultPath::new(p).ok()?;
                        let abs = vault_root.join(vp.as_str());
                        let uri = Url::from_file_path(&abs).ok()?;
                        Some(DiagnosticRelatedInformation {
                            location: Location {
                                uri,
                                range: Range::default(),
                            },
                            message: p.clone(),
                        })
                    })
                    .collect();

                diagnostics.push(Diagnostic {
                    range: doc.link_to_range(link),
                    severity: Some(DiagnosticSeverity::INFORMATION),
                    code: Some(NumberOrString::String("ambiguous-link".into())),
                    source: Some("clepsydra".into()),

                        link.target_raw,

                    ),
                    related_information: if related.is_empty() {
                        None
                    } else {
                        Some(related)
                    },
                    ..Default::default()
                });

            Some(_) => {

            }
```

**Step 1: Add `shortest_unique_prefix` to `src/lsp/rename.rs`**

```rust
/// Given a set of candidate paths, compute the shortest prefix for each
/// that uniquely distinguishes it from all others.
///
/// For each path, we walk up the path segments until the suffix is unique.
/// E.g. given `["notes/foo.md", "projects/foo.md"]`, returns
/// `["notes/foo", "projects/foo"]`.
pub fn shortest_unique_prefixes(paths: &[String]) -> Vec<String> {
    paths
        .iter()
        .map(|path| {
            let stripped = path.strip_suffix(".md").unwrap_or(path);
            let segments: Vec<&str> = stripped.split('/').collect();

            // Try increasingly long suffixes until unique
            for len in 1..=segments.len() {
                let suffix = segments[segments.len() - len..].join("/");
                let matches = paths.iter().filter(|other| {
                    let other_stripped = other.strip_suffix(".md").unwrap_or(other);
                    other_stripped.ends_with(&suffix)
                }).count();
                if matches == 1 {
                    return suffix;
                }
            }
            // Fallback: full path without .md
            stripped.to_string()
        })
        .collect()
}
```

```rust
async fn code_action(&self, params: CodeActionParams) -> Result<Option<CodeActionResponse>> {
    let uri = params.text_document.uri;
    let mut actions: Vec<CodeActionOrCommand> = Vec::new();

    // Extract link data we need while holding the documents lock
    let link_data: Vec<(crate::vault::link::Link, Range)> = {
        let docs = self.documents.lock().await;
        let doc = match docs.get(&uri) {
            Some(d) => d,
            None => return Ok(None),
        };
        doc.links
            .iter()
            .filter(|l| l.kind == crate::vault::link::LinkKind::Wiki && !(l.span.start == 0 && l.span.end == 0))
            .map(|l| (l.clone(), doc.link_to_range(l)))
            .collect()
    };

    // Get body text for raw span extraction
    let body_text = {
        let docs = self.documents.lock().await;
        docs.get(&uri).map(|d| d.body.clone())
    };
    let body_text = match body_text {
        Some(b) => b,
        None => return Ok(None),
    };

    for diag in &params.context.diagnostics {
        let code = match &diag.code {
            Some(NumberOrString::String(s)) => s.as_str(),
            _ => continue,
        };

        // Find the link matching this diagnostic range
        let link = match link_data.iter().find(|(_, range)| *range == diag.range) {
            Some((l, _)) => l,
            None => continue,
        };

        match code {
            "unresolved-link" => {
                let target = &link.target_raw;
                let new_vp = crate::vault::path::VaultPath::from_title(target);
                let new_abs = self.state.vault.resolve(&new_vp);
                let new_uri = match Url::from_file_path(&new_abs) {
                    Ok(u) => u,
                    Err(_) => continue,
                };

                let mut meta = crate::vault::page::PageMeta::new();
                meta.title = Some(target.to_string());
                let content = crate::vault::page::write_page_content(&meta, "\n");

                let mut ops: Vec<DocumentChangeOperation> = Vec::new();
                ops.push(DocumentChangeOperation::Op(ResourceOp::Create(CreateFile {
                    uri: new_uri.clone(),
                    options: Some(CreateFileOptions {
                        overwrite: Some(false),
                        ignore_if_exists: Some(true),
                    }),
                    annotation_id: None,
                })));
                ops.push(DocumentChangeOperation::Edit(TextDocumentEdit {
                    text_document: OptionalVersionedTextDocumentIdentifier {
                        uri: new_uri,
                        version: None,
                    },
                    edits: vec![OneOf::Left(TextEdit {
                        range: Range::default(),
                        new_text: content,
                    })],
                }));

                actions.push(CodeActionOrCommand::CodeAction(CodeAction {
                    title: format!("Create page: {target}"),
                    kind: Some(CodeActionKind::QUICKFIX),
                    diagnostics: Some(vec![diag.clone()]),
                    edit: Some(WorkspaceEdit {
                        changes: None,
                        document_changes: Some(DocumentChanges::Operations(ops)),
                        change_annotations: None,
                    }),
                    is_preferred: Some(true),
                    ..Default::default()
                }));
            }
            "ambiguous-link" => {
                let canonical = crate::vault::canonical::CanonicalName::from_title(&link.target_raw);
                let names = self.canonical_names.read().await;

                if let Some(candidate_paths) = names.get(canonical.as_str()) {
                    if candidate_paths.len() > 1 {
                        let prefixes = rename::shortest_unique_prefixes(candidate_paths);

                        for (path, prefix) in candidate_paths.iter().zip(prefixes.iter()) {
                            let raw_text = if link.span.end <= body_text.len() {
                                &body_text[link.span.clone()]
                            } else {
                                continue;
                            };
                            let new_text = rename::rewrite_wikilink(raw_text, prefix);

                            let edit = TextEdit {
                                range: diag.range,
                                new_text,
                            };

                            let title_display = path.strip_suffix(".md").unwrap_or(path);
                            actions.push(CodeActionOrCommand::CodeAction(CodeAction {
                                title: format!("Resolve to: {title_display}"),
                                kind: Some(CodeActionKind::QUICKFIX),
                                diagnostics: Some(vec![diag.clone()]),
                                edit: Some(WorkspaceEdit {
                                    changes: Some(
                                        [(uri.clone(), vec![edit])].into_iter().collect(),
                                    ),
                                    document_changes: None,
                                    change_annotations: None,
                                }),
                                ..Default::default()
                            }));
                        }
                    }
                }
            }
            _ => {}
        }
    }

    if actions.is_empty() {
        Ok(None)
    } else {
        Ok(Some(actions))
    }
}
```
